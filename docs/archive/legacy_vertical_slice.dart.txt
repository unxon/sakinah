class AppState extends ChangeNotifier {
  bool dark = false, showSource = true, ready = false;
  final favorites = <String>{'morning'}; final counts = <String,int>{'morning':3};
  final personal = <Map<String,String>>[];
  Future<void> load() async { final p=await SharedPreferences.getInstance(); dark=p.getBool('dark')??false; showSource=p.getBool('source')??true; ready=true; notifyListeners(); }
  Future<void> persist() async { final p=await SharedPreferences.getInstance(); await p.setBool('dark',dark); await p.setBool('source',showSource); }
  void toggleTheme(){dark=!dark; persist(); notifyListeners();} void toggleSource(){showSource=!showSource; persist(); notifyListeners();}
  void favorite(String id){favorites.contains(id)?favorites.remove(id):favorites.add(id);notifyListeners();}
  void count(Devotion d){counts[d.id]=((counts[d.id]??0)+1).clamp(0,d.count);notifyListeners();}
  void addPersonal(String title,String text){personal.add({'title':title,'text':text});notifyListeners();}
}
